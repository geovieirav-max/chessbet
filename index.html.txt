<script>
// Inicialização de Ícones
lucide.createIcons();

// Estado do Saldo do Jogador
let balance = 125.50;

// Cálculo da Comissão de 10% do Pote
const matchStakeInput = document.getElementById('matchStake');
matchStakeInput.addEventListener('input', updatePotCalculations);

function updatePotCalculations() {
const stake = parseFloat(matchStakeInput.value) || 0;
const totalPot = stake * 2;
const platformFee = totalPot * 0.10; // 10% Rake
const winnerPayout = totalPot - platformFee;

document.getElementById('calcTotalPot').innerText = `€${totalPot.toFixed(2)}`;
document.getElementById('calcPlatformFee').innerText = `€${platformFee.toFixed(2)}`;
document.getElementById('calcWinnerPayout').innerText = `€${winnerPayout.toFixed(2)}`;
}
updatePotCalculations();

// Modais
function openModal(id) {
document.getElementById(id).classList.remove('hidden');
document.getElementById(id).classList.add('flex');
}
function closeModal(id) {
document.getElementById(id).classList.add('hidden');
document.getElementById(id).classList.remove('flex');
}

// Depósito
function handleDeposit(e) {
e.preventDefault();
balance += 50.00;
document.getElementById('userBalance').innerText = `€${balance.toFixed(2)}`;
alert('Depósito de €50.00 realizado com sucesso via Cartão!');
closeModal('depositModal');
}

// Validação e Processamento de Saques (>= €50)
function validateWithdrawAmount() {
const val = parseFloat(document.getElementById('withdrawAmount').value);
const errorEl = document.getElementById('withdrawError');
const btn = document.getElementById('withdrawBtn');

if (val < 50 || isNaN(val)) {
errorEl.classList.remove('hidden');
btn.disabled = true;
btn.classList.add('opacity-50', 'cursor-not-allowed');
} else {
errorEl.classList.add('hidden');
btn.disabled = false;
btn.classList.remove('opacity-50', 'cursor-not-allowed');
}
}

function handleWithdraw(e) {
e.preventDefault();
const val = parseFloat(document.getElementById('withdrawAmount').value);

if (val > balance) {
alert('Saldo insuficiente para realizar este saque.');
return;
}

balance -= val;
document.getElementById('userBalance').innerText = `€${balance.toFixed(2)}`;
alert(`Levantamento de €${val.toFixed(2)} processado com sucesso para o seu cartão!`);
closeModal('withdrawModal');
}

// Inicialização do Tabuleiro de Xadrez (Chessboard.js)
let board = null;
let game = new Chess();

function makeRandomMove() {
const possibleMoves = game.moves();
if (possibleMoves.length === 0) return;

const randomIdx = Math.floor(Math.random() * possibleMoves.length);
game.move(possibleMoves[randomIdx]);
board.position(game.fen());
}

function onDrop(source, target) {
const move = game.move({
from: source,
to: target,
promotion: 'q'
});

if (move === null) return 'snapback';

window.setTimeout(makeRandomMove, 250);
}

function onSnapEnd() {
board.position(game.fen());
}

const config = {
draggable: true,
position: 'start',
onDrop: onDrop,
onSnapEnd: onSnapEnd
};

board = Chessboard('board', config);

function startMatch() {
game.reset();
board.start();
document.getElementById('gameStatus').innerText = 'Partida em curso! Boa sorte.';
}
